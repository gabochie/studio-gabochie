Standard breakpoints used across all pages for mobile-first responsive design.

860px

768px
Small Tablet

640px
Mobile
Hero padding reduces. Grids collapse to 1-col. Service form rows stack.
500px
Small Mobile
2-col grids collapse to 1-col. Footer collapses to 1-col.
480px
Narrow Mobile

  '  <div>Brand column</div>' +
  '  <div>Content column</div>' +
  '  <div>Support column</div>' +
  '  <div>Media column</div>' +

/* Session-aware nav auth */
function initNavAuth() {
  checkSession(function(user) {
    if (user) {
      container.innerHTML = '<a href="/dashboard/">Dashboard</a>' +
        '<a href="#" onclick="logoutUser(event)">Log Out</a>';
    } else {
      container.innerHTML = '<a href="/login/">Log In</a>' +
        '<a href="/register/" class="nav-auth-btn">Sign Up</a>';
    }
  });
}
